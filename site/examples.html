Functions

square n -> n * n
larger a b -> a > b ? a : b
add a: i32 b: i32 : i32 -> a + b

print(square(7))    // 49
print(larger(3, 8)) // 8
print(add(20, 22))  // 42

Bracket loops

total <- 0
[n <- 1..11] {
    n % 2 == 0 ? { skip }
    total += n
}
print(total) // 25

pairs <- 0
[i <- 0..4, j <- 0..i] {
    pairs += 1
}
print(pairs) // 6

Explicit mutation

increment value: &mut i64 -> {
    value += 1
}
answer = 42
count <- 41
increment(&mut count)
print(count == answer) // true

Option & Result

half n: i64 : Result[i64, i64] ->
    n % 2 == 0 ? ok(n / 2) : err(1)

half_plus_one n: i64 : Result[i64, i64] -> {
    value = ?half(n)
    ok(value + 1)
}

print(half_plus_one(40) ?? -1) // 21
print(half_plus_one(41) ?? -1) // -1

Structs & slices

struct Point { x: i64, y: i64 }
squared_length p: Point : i64 -> p.x * p.x + p.y * p.y
print(squared_length(Point { x: 3, y: 4 })) // 25

values = [10, 20, 30, 40]
middle = values[1..3]
print(len(middle)) // 2
